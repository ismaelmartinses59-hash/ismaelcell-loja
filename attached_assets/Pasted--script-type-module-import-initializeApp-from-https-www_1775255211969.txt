const app = initializeApp(firebaseConfig);

const db = getFirestore(app);

// ELEMENTOS
const linha = document.getElementById("linha");
const servico = document.getElementById("servico");
const modelo = document.getElementById("modelo");
const valor = document.getElementById("valor");
const tempo = document.getElementById("tempo");

// SERVIÇOS
const servicosPorLinha = {
  xiaomi: ["CONTA GOOGLE","CONTA MI","PAYJOY","SOFTWARE","HARD RESET"],
  samsung: ["CONTA GOOGLE","SAMSUNG CLOUD","PAYJOY","SOFTWARE","HARD RESET"],
  motorola: ["CONTA GOOGLE","MDM","SOFTWARE","HARD RESET"],

// LOGIN
window.login = async () => {
  try {
    await signInWithEmailAndPassword(auth,email.value,senha.value);

    carregar();
  } catch(e){
    alert("Erro: " + e.message);

// SERVIÇOS DINÂMICOS
linha.addEventListener("change",()=>{
  servico.innerHTML = '<option value="">Serviço</option>';
  servicosPorLinha[linha.value]?.forEach(s=>{
    servico.innerHTML += `<option>${s}</option>`;
  });
});

// TEMPO AUTOMÁTICO
servico.addEventListener("change",()=>{
  tempo.value = tempoServico[servico.value] || "";
});

// SALVAR
window.salvar = async ()=>{

  if(!auth.currentUser){
    alert("Faça login!");
    return;
  }

  if(!modelo.value || !servico.value || !valor.value){
    alert("Preencha todos os campos!");
    return;
  }

  let codigo = "OS-" + Date.now();

  await addDoc(collection(db,"precos"),{
    modelo:modelo.value,
    linha:linha.value,
    servico:servico.value,
    valor:valor.value,
    tempo:tempo.value,
    status:"aguardando",
    codigo:codigo
  });

  alert("Ordem criada: " + codigo);

  modelo.value = "";
  valor.value = "";
  tempo.value = "";

  carregar();
};

// STATUS FORMATADO
function formatarStatus(s){
  if(s === "aguardando") return "⏳ Aguardando";
  if(s === "em andamento") return "🔧 Em andamento";
  if(s === "concluido") return "✅ Concluído";
  if(s === "problema") return "⚠️ Problema";
  return s;
}

// ATUALIZAR STATUS
window.atualizarStatus = async (id, novoStatus) => {
  await updateDoc(doc(db,"precos",id),{
    status: novoStatus
  });
  carregar();
};

// CARREGAR
window.carregar = async ()=>{

  let lista = document.getElementById("lista");
  let buscaTexto = busca.value.toLowerCase();

  lista.innerHTML = "";

  const dados = await getDocs(collection(db,"precos"));

  let encontrou = false;

  dados.forEach(docSnap=>{
    let d = docSnap.data();

    let texto = (d.modelo + d.servico + d.linha).toLowerCase();

    if(buscaTexto === "" || texto.includes(buscaTexto)){

      encontrou = true;

      lista.innerHTML += `
      <div class="item">
        <strong>${d.modelo}</strong><br>
        ${d.servico} - R$ ${d.valor}<br>
        🔢 ${d.codigo}<br>
        📊 ${formatarStatus(d.status)}

        <br><br>

        <button style="background:#3b82f6" onclick="atualizarStatus('${docSnap.id}','em andamento')">Iniciar</button>
        <button style="background:#ef4444" onclick="atualizarStatus('${docSnap.id}','problema')">Problema</button>
        <button style="background:#22c55e" onclick="atualizarStatus('${docSnap.id}','concluido')">Concluir</button>

        <br><br>

        <button onclick="compartilhar(
          '${encodeURIComponent(d.modelo)}',
          '${encodeURIComponent(d.servico)}',
          '${encodeURIComponent(d.valor)}',
          '${d.codigo}'
        )">📤 Compartilhar</button>
      </div>`;
    }
  });

  lista.style.display = (buscaTexto === "" || !encontrou) ? "none":"block";
};

busca.addEventListener("input",carregar);

      <div class="linha">${servico}</div>
      <div class="linha" style="color:#0a7d4f;font-size:18px;">
        R$ ${valor}

  `;

  document.body.appendChild(card);

  });
};
